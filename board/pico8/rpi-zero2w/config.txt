# Raspberry Pi firmware configuration for the PICO-8 Zero 2 W image.
# Installed to the FAT boot partition via BR2_PACKAGE_RPI_FIRMWARE_CONFIG_FILE.
#
# start.elf is the full GPU firmware. start_cd.elf is faster to load (the
# ramdisk beta booted much quicker with it) but the cut-down blob drops
# firmware 3D; PICO-8 then dies at first GL/KMS use. Keep start_cd.elf on
# the card and swap these three lines only as an experiment:
#   start_file=start_cd.elf
#   fixup_file=fixup_cd.dat
#   gpu_mem=16

start_file=start.elf
fixup_file=fixup.dat
gpu_mem=64

kernel=zImage

# No '=' -- the firmware parser treats this line specially.
# followkernel places the ramdisk in memory immediately after zImage.
initramfs rootfs.cpio.gz followkernel

# Screen. Exactly one of these two lines is active.
#
#   hdmi.txt     the Zero 2 W's mini-HDMI port (default)
#   gpicase.txt  Retroflag GPi Case (version 1): the case's LCD on the GPIO
#                header, PWM audio to its speaker, no serial console.
#                Untested -- see the notes in that file.
#
# Each file loads vc4-kms-v3d itself.
include hdmi.txt
#include gpicase.txt

# Nothing on screen before PICO-8 (PRD 5.5, 11).
disable_splash=1
disable_overscan=1
boot_delay=0
initial_turbo=30
